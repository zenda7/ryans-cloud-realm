// === 新增：全局文本翻译辅助函数 === // 用于确保动态生成的 HTML 和搜索文本能和当前语言状态保持一致 function translateText(text) { if (!text) return text; let result = text; if (state.isTraditional) { for (let key in simToTraMap) { result = result.replace(new RegExp(key, 'g'), simToTraMap[key]); } } else { for (let key in simToTraMap) { result = result.replace(new RegExp(simToTraMap[key], 'g'), key); } } return result; } // === 修改：多语言切换函数 (复用辅助函数) === function toggleLanguage() { state.isTraditional = !state.isTraditional; document.getElementById('langTxt').innerText = state.isTraditional ? '简体' : '繁体'; const walk = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, null, false); let node; while (node = walk.nextNode()) { if (node.parentElement.tagName === 'SCRIPT' || node.parentElement.tagName === 'STYLE') continue; node.nodeValue = translateText(node.nodeValue); } } // === 修改：全局深度搜索逻辑 === function handleSearch() { const query = document.getElementById('searchInput').value.toLowerCase().trim(); const allCards = document.querySelectorAll('.list-card'); allCards.forEach(card => { // 1. 获取卡片表面的文字 let textToSearch = card.innerText; // 2. 解析 onclick 事件，深入数据源抓取详情内容 const onclickStr = card.getAttribute('onclick'); if (onclickStr) { // 匹配 selectItem('tabId', this, 'itemKey') 中的 tabId 和 itemKey const match = onclickStr.match(/selectItem\s*\(\s*['"]([^'"]+)['"]\s*,\s*(?:this|me)\s*,\s*['"]([^'"]+)['"]\s*\)/); if (match) { const tabId = match[1]; const itemKey = match[2]; let deepData = null; if (tabId === 'cards' && cardData[itemKey]) deepData = cardData[itemKey]; else if (tabId === 'prep' && prepData[itemKey]) deepData = prepData[itemKey]; else if (tabId === 'guide' && guideData[itemKey]) deepData = guideData[itemKey]; // 将深层数据转化为字符串拼接进去 if (deepData) textToSearch += " " + JSON.stringify(deepData); } } // 3. 将待匹配文本转换为当前语言（兼容繁体输入搜索），并转小写 textToSearch = translateText(textToSearch).toLowerCase(); // 4. 匹配显示逻辑 if (textToSearch.includes(query) || query === '') { card.style.display = 'flex'; } else { card.style.display = 'none'; } }); } // === 修改：渲染器：信用卡 (加入翻译处理) === function renderCards(container, list) { const htmlStr = list.map(c => `
${c.name}
${c.fee}
🎁 开卡奖励:
${c.bonus}
💳 刷卡福利:
${c.rewards}
`).join(''); // 渲染前统一进行简繁体同步 container.innerHTML = translateText(htmlStr); } // === 修改：渲染器：文章 (加入翻译处理) === function renderArticles(container, list) { const htmlStr = list.map((a, idx) => `
${a.title}

${a.fullText}
阅读全文 →
`).join(''); // 渲染前统一进行简繁体同步 container.innerHTML = translateText(htmlStr); }
